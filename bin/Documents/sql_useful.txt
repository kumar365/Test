
# Query to find salary of employees having Nth Highest salary 

> Select salary from employees order by salary DESC limit n-1,1;

# If Multiple Employees have the Same Salary:
> SELECT * FROM employee WHERE salary= (SELECT DISTINCT(salary) FROM employee ORDER BY salary DESC LIMIT n-1,1);

# Query to find salary of employees having 4th Highest salary
> Select salary from employees order by salary DESC limit 3,1;
# Here  we are skipping 3 rows from Top and returning only 1 row after skipping .

# If Multiple Employees have the Same Salary:
# Suppose you have to find 4th highest salary 
> SELECT * FROM employee WHERE salary= (SELECT DISTINCT(salary) FROM employee ORDER BY salary DESC LIMIT 3,1);


# Query to find names of employees having Nth Highest salary 

> Select emp_name from employees where salary =( Select salary from employees order by salary DESC limit n-1,1);

# Highest Salary of employee in each department
> SELECT dept_id, MAX(Salary) FROM employees GROUP BY dept_id

# Average Salary of employee in each department
> SELECT dept_id, AVG(Salary) FROM employees GROUP BY dept_id

# Query to find the employee whose salary is the highest
> select * from employee where salary=(select Max(salary) from employee);

# SQL Query for the Second-Highest Salary
> select * from employee group by salary order by  salary desc limit 1,1;

# SQL Query for the Second-Highest Salary using nested queries
> SELECT name, salary FROM employee 
WHERE salary = ( SELECT MAX(salary)  FROM employee  WHERE salary < (SELECT MAX(salary) FROM employee));
